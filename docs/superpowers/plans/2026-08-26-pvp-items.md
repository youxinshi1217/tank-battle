**Goal:** 为 PVP 纯 1v1 启用道具刷新与双方竞争拾取，并将道具系统全局调优（刷新间隔 5→3s、同场上限 1→2、在场时长 15→20s）应用于全模式。

```python
class ItemSpawnManagerEx(ItemSpawnManager):
    """道具刷新增强（T7）：以 settings 的 ITEM_LIFETIME 创建道具，覆盖冻结文件默认 15s。

    不改冻结文件 item_spawn_manager.py；刷新间隔/数量经构造参数传入（main._load_stage）。
    多玩家避让：PVP/PvPvE 中把 P2 的 rect 并入 update() 的 enemy_rects 列表即可。
    """

    def _spawn_one(self, player_rect, enemy_rects):
        pos = self._find_spawn_position(player_rect, enemy_rects)
        if pos is None:
            return None
        cx, cy = pos
        item = Item(cx, cy, self._choose_type(), lifetime=ITEM_LIFETIME)
        self._bind_density(item)
        return item
```

```python
# -*- coding: utf-8 -*-
"""Task7 验证：PVP 启用道具（竞争拾取）+ 道具系统全局调优（间隔3s/同场2/寿命20s）。"""
import os, sys
os.environ.setdefault("SDL_VIDEODRIVER", "dummy")
os.environ.setdefault("SDL_AUDIODRIVER", "dummy")
sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
import pygame
pygame.init()

import main as M
from core.settings import ITEM_SPAWN_INTERVAL, ITEM_MAX_ACTIVE, ITEM_LIFETIME
from entities.item import Item, ItemType

# 1) 全局调优参数已写入 settings（唯一事实来源）
assert ITEM_SPAWN_INTERVAL == 3.0, ITEM_SPAWN_INTERVAL
assert ITEM_MAX_ACTIVE == 2, ITEM_MAX_ACTIVE
assert ITEM_LIFETIME == 20.0, ITEM_LIFETIME

game = M.Game()

def step(frames, fps=60):
    for _ in range(frames):
        game.clock.tick(fps)
        game.update()

# 2) solo：构造即用增强类 + 新参数（间隔/数量）
game.start_game(0, "简单")
assert isinstance(game.item_spawn_manager, M.ItemSpawnManagerEx), type(game.item_spawn_manager)
assert game.item_spawn_manager.spawn_interval == 3.0
assert game.item_spawn_manager.max_active == 2

# 3) PVP：道具刷新开启、仍无敌人/无波次
game.start_pvp(rounds=5)
game.round_over_pause = 0.0
game._pause_after = None
game._pvp_destroyed = {}
step(240)   # 4s > 3s 间隔，应至少刷新 1 个道具
assert len(game.enemy_tanks) == 0, "PVP 应无敌人"
assert len(game.item_spawn_manager.items) >= 1, "PVP 应刷新道具"

# 4) 现存道具寿命为 20s
for it in game.item_spawn_manager.items:
    assert abs(it.lifetime - 20.0) < 1e-6, f"寿命应 20s，实际 {it.lifetime}"

# 5) P1 拾取接线：道具放 P1 脚下，update 一帧被拾取
game.item_spawn_manager.items = []
p1c = game.player_tank.rect.center
it1 = Item(p1c[0], p1c[1], ItemType.SPEED, lifetime=20.0)
game.item_spawn_manager.items = [it1]
step(2)
assert it1 not in game.item_spawn_manager.items, "P1 应拾取脚下道具"

# 6) P2 拾取接线：道具放 P2 脚下，update 一帧被 P2 拾取
p2c = game.player2_tank.rect.center
it2 = Item(p2c[0], p2c[1], ItemType.SHIELD, lifetime=20.0)
game.item_spawn_manager.items = [it2]
step(2)
assert it2 not in game.item_spawn_manager.items, "P2 应拾取脚下道具"

# 7) P2 效果计时：update() 驱动 P2 的 update_effects，护盾递减
game.player2_tank.shield_remaining = 10.0
step(60)   # 1s
assert game.player2_tank.shield_remaining < 10.0, "P2 护盾应随时间递减"

print("T7 OK")
```
